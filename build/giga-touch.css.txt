/* ---------- Phone and tablet ----------
   Measured with tools/g-responsive.cjs across 360 / 390 / 414 / 768 / 820 /
   1024 / 1180. Three things: nothing interactive below ~44px, no text below
   12px, and the layouts that had room to do better on a tablet doing so. */

/* In a column flex container `flex:1` sets flex-basis:0 on the BLOCK axis,
   which overrode height:52px and collapsed the stacked postcode field to
   19px — on every page's closing CTA below 560px. */
@media (max-width:560px){
  .g .g-form input { flex:none; width:100%; }
}

/* Tablet portrait has room for two testimonials and two feature cards
   abreast; one column wasted most of the width. */
@media (min-width:640px) and (max-width:859px){
  .g .g-quotes { grid-template-columns:repeat(2,1fr); }
}
